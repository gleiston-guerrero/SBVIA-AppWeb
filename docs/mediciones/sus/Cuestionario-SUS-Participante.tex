\seccion{Instrucciones para el participante}

\begin{enumerate}[label=\textbf{\arabic*.}, leftmargin=5.6mm, labelsep=1.8mm,
                  itemsep=1.4pt, topsep=0pt, parsep=0pt]
  \item \small Escribe \textbf{\'unicamente tu c\'odigo de participante} en la casilla de
        abajo. No escribas tu nombre ni tus apellidos.
  \item \small Responde este cuestionario \textbf{inmediatamente despu\'es} de haber usado
        el sistema SBVIA.
  \item \small Marca \textbf{una sola casilla por pregunta}. No hay respuestas correctas
        ni incorrectas.
  \item \small Responde seg\'un lo que \textbf{realmente sentiste} al usar el sistema,
        sin pensarlo demasiado.
  \item \small \textbf{No dejes ninguna pregunta sin responder.}
\end{enumerate}
